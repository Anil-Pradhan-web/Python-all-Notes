# Q10. Profile and Media in content_project

## Step 1: Create Profile Model (`models.py`)

```python
class Profile(models.Model):
    user = models.OneToOneField(
        User,
        on_delete=models.CASCADE
    )

    image = models.ImageField(
        upload_to='profiles/'
    )
```

## Step 2: Create Edit Form (`forms.py`)

```python
class ProfileEditForm(forms.ModelForm):

    class Meta:
        model = Profile
        fields = ['image']
```

## Step 3: Create Edit View (`views.py`)

```python
@login_required
def edit(request):

    if request.method == 'POST':
        profile_form = ProfileEditForm(
            request.POST,
            request.FILES,
            instance=request.user.profile
        )

        if profile_form.is_valid():
            profile_form.save()

    else:
        profile_form = ProfileEditForm(
            instance=request.user.profile
        )

    return render(
        request,
        'account/edit.html',
        {'profile_form': profile_form}
    )
```
